void reverse(int arr[], int size) {
int mid = size / 2;
for (int i = 0; i < mid; i++) {
const int temp = arr[i];
arr[i] = arr[size - i - 1];
arr[size - i - 1] = temp;
}
}

int start = 0;
int end = size - 1;
while (start < end) {
swap(arr[start], arr[end]);
start++;
end--;
}
}

void reverse(int arr[], int start, int end) {
if (start >= end) return;
std::swap(arr[start], arr[end]);
reverse(arr, start + 1, end - 1);
}
